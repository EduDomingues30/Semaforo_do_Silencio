/*
  Calibração do sensor de som

  Este sketch não acende LED nenhum. Ele apenas mostra a leitura
  bruta do sensor no Serial Monitor, para que a turma descubra
  quais números correspondem ao silêncio, à conversa normal e ao
  barulho da sala inteira falando.

  Como usar
    1. Envie este código para a placa
    2. Abra o Serial Monitor em 9600 baud
    3. Peça silêncio total e anote a faixa de valores
    4. Peça conversa normal e anote
    5. Peça que todos falem ao mesmo tempo e anote
    6. Escolha os limiares a partir do que foi observado
*/

const int PINO_SENSOR = A0;

void setup() {
  Serial.begin(9600);
  Serial.println("Calibracao iniciada");
  Serial.println("Anote os valores em cada situacao");
}

void loop() {
  int leitura = analogRead(PINO_SENSOR);
  Serial.println(leitura);
  delay(200);
}
